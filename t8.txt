(* Programa para calcular a media de numeros positivos *)
program media;
var
  n, i, num, soma, media : integer;
begin
  read(n);
  soma := 0;
  i := 0;
  while i < n do
  begin
    read(num);
    if num > 0 then
    begin
      soma := soma + num;
      i := i + 1
    end
  end;
  if i > 0 then
    media := soma div i
  else
    media := 0;
  write(media)
end.
